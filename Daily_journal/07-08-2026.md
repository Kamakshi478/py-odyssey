# Dictionaries 

## Creating dictionary 
     profile={"name": "John Doe", "age": 30, "Gender": "Male"}
     print(profile)

## Accessing elements
    profile={"name": "John Doe", "age": 30, "Gender": "Male"}
    print(profile["name"])

## Adding

       profile={"name": "John Doe", "age": 30, "Gender": "Male"}
       profile["Occupation"]="Software Engineer"
       print(profile)

## Updating

     profile={'name': 'John Doe', 'age': 30, 'Gender': 'Male', 'Occupation': 'Software Engineer'}
     profile["Occupation"]="Bank manager"
     print(profile)

## Deleting

    profile={"name": "John Doe", "age": 30, "Gender": "Male"}
    profile.pop("name")
    print(profile)


# Lists 
     ## Creation 
     Fruits= ["apple", "banana", "cherry"]
     print(Fruits)

## Accessing 

    Fruits= ["apple", "banana", "cherry"]
    print(Fruits[0])
    print(Fruits[1])
    print(Fruits[2])

## Modifying 

    Fruits= ["apple", "banana", "cherry"]
    Fruits[1]="Mango"
    print(Fruits)

## Adding Elements 
    ### By using .append()

    Fruits= ["apple", "banana", "cherry"]
    Fruits.append("orange") # Adds orange to the end of the list
    print(Fruits)

    ### By using .insert()

    Fruits= ["apple", "banana", "cherry"]
    Fruits.insert(2,"orange") 
    print(Fruits)

## Removing Element 

    Fruits= ["apple", "banana", "cherry"]
    Fruits.remove("banana")
    print(Fruits)
